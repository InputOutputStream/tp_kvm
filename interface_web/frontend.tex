\section{Introduction}

\subsection{Contexte du Projet}

THOTH CLOUD est une plateforme de gestion cloud complète développée dans le cadre du TP4 d'INF4107. Le projet vise à créer un mini cloud offrant trois niveaux de services :

\begin{itemize}[leftmargin=*]
    \item \textbf{IaaS} (Infrastructure as a Service) : Provisionnement et gestion de machines virtuelles
    \item \textbf{PaaS} (Platform as a Service) : Déploiement d'applications conteneurisées via Docker Swarm
    \item \textbf{SaaS} (Software as a Service) : Plateforme collaborative de type Google Workspace
\end{itemize}

\subsection{Objectifs Frontend}

Ce rapport se concentre sur l'implémentation de l'interface web moderne et intuitive permettant :

\begin{successbox}[Fonctionnalités Principales]
\begin{itemize}
    \item Gestion multi-utilisateurs avec isolation complète des ressources
    \item Interface responsive compatible desktop/mobile/tablette
    \item Tableau de bord temps réel avec monitoring des ressources
    \item Catalogue d'applications PaaS déployables en un clic
    \item Gestion de clusters Docker Swarm à la demande
    \item Système de facturation intégré
    \item Thème clair/sombre avec préférences utilisateur
\end{itemize}
\end{successbox}

Implémentation d'un système d'authentification complet avec JWT :

\begin{lstlisting}[language=JavaScript,caption=Service d'authentification (auth.js)]
class AuthService {
    constructor() {
        this.token = localStorage.getItem('thoth_token');
        this.currentUser = JSON.parse(
            localStorage.getItem('thoth_user')
        );
    }
    
    async login(username, password) {
        const response = await fetch(`${API_URL}/auth/login`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ username, password })
        });
        
        const data = await response.json();
        
        if (data.success) {
            this.token = data.token;
            this.currentUser = data.user;
            localStorage.setItem('thoth_token', data.token);
            localStorage.setItem('thoth_user', 
                JSON.stringify(data.user));
            return { success: true };
        }
        return { success: false, error: data.error };
    }
    
    isAdmin() {
        return this.currentUser?.role === 'admin';
    }
    
    logout() {
        this.token = null;
        this.currentUser = null;
        localStorage.clear();
        window.location.href = 'login.html';
    }
}
\end{lstlisting}

\subsection{Découplage de l'Application (Exigence 2)}

\subsubsection{Configuration API Dynamique}

L'application frontend est complètement découplée :

\begin{lstlisting}[language=JavaScript,caption=Configuration API (api-service.js)]
class ApiService {
    constructor() {
        // URL configurable via variable d'environnement
        this.baseURL = window.API_URL || 
                       'http://localhost:3000/api';
    }
    
    async request(endpoint, options = {}) {
        const url = `${this.baseURL}${endpoint}`;
        
        // Ajout automatique du token
        const headers = {
            'Content-Type': 'application/json',
            ...options.headers
        };
        
        if (window.authService?.token) {
            headers['Authorization'] = 
                `Bearer ${window.authService.token}`;
        }
        
        const response = await fetch(url, {
            ...options,
            headers
        });
        
        return response.json();
    }
}
\end{lstlisting}

\begin{infobox}[Avantages du Découplage]
\begin{itemize}
    \item Frontend déployable sur n'importe quel serveur web statique
    \item Backend containerisé accessible via API REST
    \item Communication via réseau (local ou distant)
    \item Scalabilité indépendante frontend/backend
\end{itemize}
\end{infobox}

function generateAppCompose(app, username, dbCredentials) {
    const appId = `${username}_${app.id}`;
    let environment = {};
    
    // Configuration selon le type de DB
    if (dbCredentials) {
        if (app.database === 'mariadb') {
            environment = {
                MYSQL_HOST: dbCredentials.host,
                MYSQL_DATABASE: dbCredentials.database,
                MYSQL_USER: dbCredentials.user,
                MYSQL_PASSWORD: dbCredentials.password,
                MYSQL_PORT: dbCredentials.port.toString()
            };
        } else if (app.database === 'postgresql') {
            environment = {
                POSTGRES_HOST: dbCredentials.host,
                POSTGRES_DB: dbCredentials.database,
                POSTGRES_USER: dbCredentials.user,
                POSTGRES_PASSWORD: dbCredentials.password,
                POSTGRES_PORT: dbCredentials.port.toString()
            };
        }
    }
    
    // Génération YAML
    const yaml = `
version: '3.8'
services:
  ${appId}:
    image: ${app.image}
    container_name: ${appId}
    restart: unless-stopped
    ports:
${app.ports.map(p => `      - "${p}"`).join('\n')}
    environment:
${Object.entries(environment).map(([k, v]) => 
    `      ${k}: "${v}"`).join('\n')}
    networks:
      - thoth_paas_network
    volumes:
      - ${appId}_data:/var/www/html

volumes:
  ${appId}_data:

networks:
  thoth_paas_network:
    external: true
`;
    
    return { environment, yaml };
}

class SaaSBillingService {
    calculateStorageCost(usedGB) {
        const tiers = [
            { limit: 10, price: 0 },        // 10 GB gratuits
            { limit: 50, price: 1000 },     // 1000 FCFA/10GB
            { limit: 100, price: 1500 },    // 1500 FCFA/50GB
            { limit: Infinity, price: 2000 } // 2000 FCFA/100GB+
        ];
        
        let cost = 0;
        let remainingGB = usedGB;
        
        for (const tier of tiers) {
            if (remainingGB <= 0) break;
            
            const tierGB = Math.min(
                remainingGB, 
                tier.limit - (usedGB - remainingGB)
            );
            cost += (tierGB / 10) * tier.price;
            remainingGB -= tierGB;
        }
        
        return {
            usedGB: usedGB,
            cost: cost.toFixed(2),
            tier: this.getCurrentTier(usedGB)
        };
    }
}

\begin{lstlisting}[language=JavaScript,caption=Gestion d'état (state-manager.js)]
class StateManager {
    constructor() {
        this.state = {
            hosts: [],
            vms: [],
            paasApps: [],
            networks: [],
            users: [],
            loading: {},
            errors: {},
            lastUpdated: {}
        };
        this.listeners = new Map();
        this.cache = new Map();
        this.cacheTimeout = 30000; // 30 secondes
    }
    
    // Souscription aux changements
    subscribe(key, callback) {
        if (!this.listeners.has(key)) {
            this.listeners.set(key, new Set());
        }
        this.listeners.get(key).add(callback);
        
        return () => {
            this.listeners.get(key)?.delete(callback);
        };
    }
    
    // Mise à jour d'état avec notification
    setState(key, value) {
        this.state[key] = value;
        this.state.lastUpdated[key] = Date.now();
        this.notify(key, value);
    }
    
    // Retry avec backoff exponentiel
    async withRetry(fn, context, maxRetries = 3, baseDelay = 1000) {
        for (let attempt = 1; attempt <= maxRetries; attempt++) {
            try {
                this.setLoading(context, true);
                const result = await fn();
                this.setLoading(context, false);
                return result;
            } catch (error) {
                if (attempt === maxRetries) {
                    this.setError(context, error);
                    throw error;
                }
                const delay = baseDelay * Math.pow(2, attempt - 1);
                await this.sleep(delay);
            }
        }
    }
}

class ThemeManager {
    constructor() {
        this.theme = localStorage.getItem('thoth_theme') || 'light';
        this.init();
    }

    init() {
        this.applyTheme(this.theme);
        
        // Écoute des changements système
        const prefersDark = window.matchMedia(
            '(prefers-color-scheme: dark)'
        );
        prefersDark.addEventListener('change', (e) => {
            if (localStorage.getItem('thoth_theme') === 'system') {
                this.applyTheme(e.matches ? 'dark' : 'light');
            }
        });
    }

    applyTheme(theme) {
        const html = document.documentElement;
        
        if (theme === 'dark' || 
            (theme === 'system' && 
             window.matchMedia('(prefers-color-scheme: dark)').matches)) {
            html.classList.add('dark-theme');
            this.updateThemeIcon('dark');
        } else {
            html.classList.remove('dark-theme');
            this.updateThemeIcon('light');
        }
    }

    toggleTheme() {
        const themes = ['light', 'dark', 'system'];
        let currentIndex = themes.indexOf(this.theme);
        this.theme = themes[(currentIndex + 1) % themes.length];
        
        localStorage.setItem('thoth_theme', this.theme);
        this.applyTheme(this.theme);
        
        showToast(`�� Thème ${this.theme} activé`, 'info');
    }
}

function validatePassword(password) {
    const minLength = 8;
    const hasUpperCase = /[A-Z]/.test(password);
    const hasLowerCase = /[a-z]/.test(password);
    const hasNumbers = /\d/.test(password);
    const hasSpecialChar = /[!@#$%^&*]/.test(password);
    
    if (password.length < minLength) {
        return { valid: false, error: 'Minimum 8 caractères' };
    }
    
    if (!hasUpperCase || !hasLowerCase) {
        return { valid: false, error: 'Majuscules et minuscules requises' };
    }
    
    if (!hasNumbers) {
        return { valid: false, error: 'Au moins un chiffre requis' };
    }
    
    return { valid: true };
}

function sanitizeInput(input) {
    const div = document.createElement('div');
    div.textContent = input;
    return div.innerHTML;
}

class ApiService {
    async request(endpoint, options = {}) {
        const headers = {
            'Content-Type': 'application/json',
            ...options.headers
        };
        
        // Ajout token CSRF
        const csrfToken = this.getCsrfToken();

\subsubsection{Cache Côté Client}

\begin{lstlisting}[language=JavaScript,caption=Système de cache (state-manager.js)]
class StateManager {
    setCache(key, data, ttl = 30000) {
        const cacheEntry = {
            data,
            timestamp: Date.now(),
            ttl
        };
        this.cache.set(key, cacheEntry);
    }

    getCache(key) {
        const entry = this.cache.get(key);
        if (!entry) return null;
        
        if (Date.now() - entry.timestamp > entry.ttl) {
            this.cache.delete(key);
            return null;
        }
        
        return entry.data;
    }
}

// Utilisation
async function loadVMs() {
    const cached = window.stateManager.getCache('vms');
    if (cached) {
        renderVMs(cached);
        return;
    }
    
    const data = await window.apiService.getVMs();
    window.stateManager.setCache('vms', data.vms);
    renderVMs(data.vms);
}
\end{lstlisting}

\subsubsection{Debouncing et Throttling}

\begin{lstlisting}[language=JavaScript,caption=Debounce (navigation.js)]
function debounce(func, wait) {
    let timeout;
    return function executedFunction(...args) {
        const later = () => {
            clearTimeout(timeout);
            func(...args);
        };
        clearTimeout(timeout);
        timeout = setTimeout(later, wait);
    };
}

// Utilisation pour recherche
vmSearchInput.addEventListener('input', debounce((e) => {
    const searchTerm = e.target.value.toLowerCase();
    filterVMs(searchTerm);
}, 300));

\section{Limites et Améliorations Futures}

\subsection{Limites Connues}

\begin{warningbox}[Limitations Actuelles]
\begin{itemize}
    \item \textbf{Temps Réel} : Pas de WebSocket pour le monitoring live (polling toutes les 3s)
    \item \textbf{Offline} : Pas de support mode hors ligne ou PWA
    \item \textbf{i18n} : Interface uniquement en français, pas de multilingue
    \item \textbf{Graphiques} : Visualisations limitées, pas de bibliothèque de charts
    \item \textbf{Tests} : Absence de tests automatisés (unit tests, e2e)
    \item \textbf{Bundle} : Pas de minification/bundling (webpack, vite)
    \item \textbf{Mobile} : Interface responsive mais expérience mobile perfectible
\end{itemize}
\end{warningbox}

\subsection{Améliorations Proposées}

\begin{successbox}[Roadmap Future]
\begin{enumerate}
    \item \textbf{Court Terme (1-2 mois)}

        \item Internationalisation (i18n)
